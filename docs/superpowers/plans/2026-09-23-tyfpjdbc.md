# TyFPJDBC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 Lazarus / FPC 64bit 落地 JNI 到 JDBC 桥，跑通只读展示、连接池并发、可编辑 Dataset、驱动下载加裁剪 JRE 工具链。

**Architecture:** FPC App 经 TyFPJDBC.Core 调 JNI，只调薄门面 tyfpjdbc-bridge.jar 的十几个扁平接口，批取、池化、类型展开在 Java 侧做，DB 层做命名参数转换和 TBufDataset 适配，LCL 只做设计期对话框。

**Tech Stack:** Free Pascal 3.2.2 / Lazarus 3.x 64bit, Java 25 LTS Temurin, HikariCP 5.x, slf4j-api, Gradle 8.x, FPCUnit, H2 2.x, SQLite xerial, PostgreSQL 17 原生版。

**Spec:** `docs/superpowers/specs/2026-09-23-tyfpjdbc-design.md`

## Global Constraints

- 只支持 64bit 应用，不支持 Windows 32bit，FPC 程序与 JVM 位数必须一致。
- TyFPJDBC.Core 只 uses SysUtils, Classes, syncobjs，禁止引用 Forms, Dialogs, LCL，CI 用 grep Forms 卡点。
- 驱动 jar 不进 jlink 镜像，镜像里只有 JDK 模块加 bridge 加 HikariCP，drivers/ 目录运行时用 URLClassLoader 动态加载。
- 所有取数走 FetchBatch 列式批量，默认 1000 行，一个格子调一次 JNI 的写法禁用，大 BLOB 走 InputStream 到 TStream 流式。
- 所有值一律走参数绑定，禁止字符串拼接拼值，Macro 只拼标识符且走白名单。
- Pool 全局共享，Connection / Query 按请求独占，TDataset 不跨线程，工作线程首次自动 AttachCurrentThread，结束 Detach。
- jstring 来回转统一 UTF-8，JVM 启动必带 -Dfile.encoding=UTF-8 -Djava.awt.headless=true，服务端加 -Xrs。
- 连接时先对 GetBridgeVersion，jar 和 Pascal 单元对不上直接抛错。

```pascal
uses SysUtils, Classes, syncobjs;
type
  TJVMManager = class
  private
    class var FLock: TCriticalSection;
    class var FStarted: Boolean;
  public
    class constructor Create;
    class destructor Destroy;
    class function IsHeadlessArg(const S: string): Boolean; static;
    class function BuildDesktopArgs: string; static;
    class function FindLibJvm(const CustomPath: string): string; static;
    class procedure EnsureStarted(const LibJvm, ExtraArgs: string); static;
    class procedure AttachThread; static;
    class procedure DetachThread; static;
  end;
implementation
class constructor TJVMManager.Create;
begin FLock := TCriticalSection.Create; end;
class destructor TJVMManager.Destroy;
begin FLock.Free; end;
class function TJVMManager.IsHeadlessArg(const S: string): Boolean;
begin Result := Pos('headless=true', S) > 0; end;
class function TJVMManager.BuildDesktopArgs: string;
begin Result := '-Xmx512m -Dfile.encoding=UTF-8 -Djava.awt.headless=true'; end;
class function TJVMManager.FindLibJvm(const CustomPath: string): string;
var h: string;
begin
  if (CustomPath <> '') and FileExists(CustomPath) then Exit(CustomPath);
  h := GetEnvironmentVariable('JAVA_HOME');
  Result := IncludeTrailingPathDelimiter(h) + 'bin/server/jvm.dll';
  if not FileExists(Result) then
    raise Exception.Create('libjvm not found, set JAVA_HOME or bundle jre/');
end;
class procedure TJVMManager.EnsureStarted(const LibJvm, ExtraArgs: string);
begin
  FLock.Enter; try
    if FStarted then Exit;
    if Pos('headless=true', ExtraArgs) = 0 then
      raise Exception.Create('must pass headless=true');
    FStarted := True;
  finally FLock.Leave; end;
end;
class procedure TJVMManager.AttachThread; begin end;
class procedure TJVMManager.DetachThread; begin end;
end.
```

```pascal
unit TyFPJDBC.Sql.Parser;
{$mode objfpc}{$H+}
interface
uses SysUtils, Classes;
type
  TSqlParseResult = record JdbcSql: string; ParamOrder: array of string; end;
  TSqlParser = class
    class function Parse(const SQL: string): TSqlParseResult; static;
    class function ExpandMacro(const SQL, Name, Value: string): string; static;
    class function CheckMacro(const Value: string): Boolean; static;
  end;
implementation
uses TyFPJDBC.JNI.Utils;
class function TSqlParser.CheckMacro(const Value: string): Boolean;
begin Result := TJniUtils.IsWhiteIdent(Value); end;
class function TSqlParser.ExpandMacro(const SQL, Name, Value: string): string;
begin
  if not CheckMacro(Value) then raise Exception.Create('bad macro: ' + Value);
  Result := StringReplace(SQL, '&' + Name, Value, [rfReplaceAll]);
end;
class function TSqlParser.Parse(const SQL: string): TSqlParseResult;
var i, n: Integer; ch: Char; inStr: Boolean; q: Char; outp, nm: string;
begin
  outp := ''; SetLength(Result.ParamOrder, 0);
  i := 1; n := Length(SQL); inStr := False; q := #0;
  while i <= n do begin
    ch := SQL[i];
    if inStr then begin
      outp += ch;
      if ch = q then inStr := False;
      Inc(i); Continue;
    end;
    if (ch = '''') or (ch = '"') then begin inStr := True; q := ch; outp += ch; Inc(i); Continue; end;
    if (ch = '-') and (i < n) and (SQL[i+1] = '-') then begin
      while (i <= n) and (SQL[i] <> #10) do begin outp += SQL[i]; Inc(i); end; Continue;
    end;
    if (ch = ':') and (i < n) and (SQL[i+1] = ':') then begin outp += '::'; Inc(i, 2); Continue; end;
    if (ch = ':') and (i < n) and (SQL[i+1] in ['A'..'Z','a'..'z','_']) then begin
      nm := ''; Inc(i);
      while (i <= n) and (SQL[i] in ['A'..'Z','a'..'z','0'..'9','_']) do begin nm += SQL[i]; Inc(i); end;
      outp += '?';
      SetLength(Result.ParamOrder, Length(Result.ParamOrder)+1);
      Result.ParamOrder[High(Result.ParamOrder)] := nm;
      Continue;
    end;
    outp += ch; Inc(i);
  end;
  Result.JdbcSql := outp;
end;
end.
```

```pascal
unit TyFPJDBC.Type.Map;
{$mode objfpc}{$H+}
interface
uses SysUtils, DB;
type TJdbcTypeMap = class
  class function ToFieldType(const J: string): TFieldType; static;
  class function NeedStream(const J: string): Boolean; static;
end;
implementation
class function TJdbcTypeMap.ToFieldType(const J: string): TFieldType;
var u: string;
begin
  u := UpperCase(Trim(J));
  if (u='VARCHAR') or (u='NVARCHAR') or (u='CHAR') then Exit(ftWideString);
  if (u='CLOB') or (u='NCLOB') or (u='JSON') or (u='JSONB') or (u='UUID') or (u='SQLXML') then Exit(ftWideMemo);
  if (u='INTEGER') or (u='SMALLINT') then Exit(ftInteger);
  if u='BIGINT' then Exit(ftLargeint);
  if (u='NUMERIC') or (u='DECIMAL') then Exit(ftFmtBCD);
  if (u='FLOAT') or (u='DOUBLE') or (u='REAL') then Exit(ftFloat);
  if (u='BOOLEAN') or (u='BIT') then Exit(ftBoolean);
  if u='DATE' then Exit(ftDate);
  if u='TIME' then Exit(ftTime);
  if Pos('TIMESTAMP', u) = 1 then Exit(ftDateTime);
  if (u='BLOB') or (u='BYTEA') or (u='BINARY') or (u='VARBINARY') then Exit(ftBlob);
  if (u='ARRAY') or (u='STRUCT') then Exit(ftWideMemo);
  Result := ftWideString;
end;
class function TJdbcTypeMap.NeedStream(const J: string): Boolean;
begin Result := ToFieldType(J) = ftBlob; end;
end.
```

```pascal
unit TyFPJDBC.Connection;
{$mode objfpc}{$H+}
interface
uses SysUtils, Classes;
type
  EJDBCError = class(Exception)
    SQLState: string; VendorCode: Integer; Chain: TStringList;
    constructor CreateChain(const Msg, State: string; Code: Integer; const Next: string);
    destructor Destroy; override;
  end;
  TJDBCConnection = class
    AutoCommit: Boolean;
    procedure Borrow; virtual;
    procedure Release; virtual;
    procedure Commit; virtual;
    procedure Rollback; virtual;
  end;
implementation
constructor EJDBCError.CreateChain(const Msg, State: string; Code: Integer; const Next: string);
begin inherited Create(Msg + ' caused by ' + Next); SQLState := State; VendorCode := Code;
  Chain := TStringList.Create; Chain.Add(Msg); Chain.Add(Next); end;
destructor EJDBCError.Destroy; begin Chain.Free; inherited; end;
procedure TJDBCConnection.Borrow; begin end;
procedure TJDBCConnection.Release; begin end;
procedure TJDBCConnection.Commit; begin if AutoCommit then raise EJDBCError.CreateChain('autocommit on','','0','commit not allowed'); end;
procedure TJDBCConnection.Rollback; begin end;
end.
```
